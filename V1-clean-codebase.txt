import React, { useState, useRef, useEffect, useCallback, useMemo } from 'react';
import {
  ROW_HEIGHT,
  CANVAS_PAD_Y,
  formatISODate,
  generateTimelineDays,
  generateSafeId,
  sanitizeTask,
  sanitizeHabit,
  loadPersistedTasks,
  loadPersistedHabits,
  loadPersistedRevisions,
  persistState,
  executeMatrixAction,
  toggleHabitToday,
  renameTask,
  renameHabit,
  updateTaskColor,
  moveTaskPosition,
  insertNewTask,
  insertNewHabit,
} from './trackerStore';

import { Header, FooterControls } from './components/Header';
import { TimelineStream } from './components/TimelineStream';
import { TaskModal, HabitModal, NameHistoryModal } from './components/Modals';
import { ImportInspectorModal, RevisionsModal, VaultDrawer } from './components/BackupVaultModals';

export default function App() {
  const [tasks, setTasks] = useState(loadPersistedTasks);
  const [habits, setHabits] = useState(loadPersistedHabits);
  const [revisions, setRevisions] = useState(loadPersistedRevisions);
  const [selectedTaskId, setSelectedTaskId] = useState(null);

  // Sync to real laptop time
  const [now, setNow] = useState(() => new Date());
  const todayIso = useMemo(() => formatISODate(now), [now]);

  // Generate dynamic calendar slice: 21 days back, 7 days forward
  // Adaptively spans as many months back as required (minimum 90 days)
  const timelineDays = useMemo(() => generateTimelineDays(now, tasks, habits), [now, tasks, habits]);

  // Today's day index in the current slice
  const todayIndex = useMemo(() => {
    const idx = timelineDays.findIndex((d) => d.isoDate === todayIso);
    return idx !== -1 ? idx : 21;
  }, [timelineDays, todayIso]);

  // Refresh clock every minute to detect midnight turnover automatically
  useEffect(() => {
    const timer = setInterval(() => {
      setNow(new Date());
    }, 60000);
    return () => clearInterval(timer);
  }, []);

  const [taskModalOpen, setTaskModalOpen] = useState(false);
  const [taskForm, setTaskForm] = useState({ title: '', hex: '#34d399', position: 1 });

  const [habitModalOpen, setHabitModalOpen] = useState(false);
  const [habitForm, setHabitForm] = useState({ label: '', hex: '#22d3ee', position: 1 });

  const [vaultOpen, setVaultOpen] = useState(false);
  const [historyModalTask, setHistoryModalTask] = useState(null);
  const [revisionsModalOpen, setRevisionsModalOpen] = useState(false);

  const [importFilePayload, setImportFilePayload] = useState(null);
  const [selectedImportTaskIds, setSelectedImportTaskIds] = useState(new Set());

  const [purgeConfirmId, setPurgeConfirmId] = useState(null);
  const [purgeInputText, setPurgeInputText] = useState('');

  const [editingItemId, setEditingItemId] = useState(null);
  const [editValue, setEditValue] = useState('');

  const scrollContainerRef = useRef(null);
  const fileInputRef = useRef(null);

  const snapshotState = (reason = 'Manual Change') => {
    const snap = {
      id: generateSafeId('rev'),
      at: new Date().toLocaleTimeString(),
      reason,
      tasks: JSON.parse(JSON.stringify(tasks)),
      habits: JSON.parse(JSON.stringify(habits)),
    };
    setRevisions((prev) => [snap, ...prev.slice(0, 19)]);
  };

  useEffect(() => {
    persistState(tasks, habits, revisions);
  }, [tasks, habits, revisions]);

  const centerCurrentWeek = (smooth = true) => {
    if (!scrollContainerRef.current) return;
    const targetScroll = CANVAS_PAD_Y + Math.max(0, todayIndex - 3) * ROW_HEIGHT - 12;
    scrollContainerRef.current.scrollTo({
      top: Math.max(0, targetScroll),
      behavior: smooth ? 'smooth' : 'auto',
    });
  };

  useEffect(() => {
    centerCurrentWeek(false);
  }, [todayIndex]);

  const activeTasks = tasks.filter((t) => !t.isArchived);
  const archivedTasks = tasks.filter((t) => t.isArchived);

  const handleCreateTask = (e) => {
    e.preventDefault();
    if (!taskForm.title.trim()) return;
    snapshotState(`Spawn Task: ${taskForm.title}`);
    setTasks((prev) =>
      insertNewTask(prev, { title: taskForm.title, hex: taskForm.hex, position: parseInt(taskForm.position, 10) })
    );
    setTaskModalOpen(false);
  };

  const handleCreateHabit = (e) => {
    e.preventDefault();
    if (!habitForm.label.trim()) return;
    snapshotState(`Spawn Habit: ${habitForm.label}`);
    setHabits((prev) =>
      insertNewHabit(prev, { label: habitForm.label, hex: habitForm.hex, position: parseInt(habitForm.position, 10) })
    );
    setHabitModalOpen(false);
  };

  const handleArchiveTask = useCallback((taskId) => {
    snapshotState('Archive Task');
    setTasks((prev) => prev.map((t) => (t.id === taskId ? { ...t, isArchived: true } : t)));
    setSelectedTaskId((prev) => (prev === taskId ? null : prev));
  }, [tasks]);

  const handleRestoreTask = (taskId) => {
    snapshotState('Respawn Task from Vault');
    setTasks((prev) => prev.map((t) => (t.id === taskId ? { ...t, isArchived: false } : t)));
  };

  const handleHardPurgeTask = (taskId) => {
    if (purgeInputText.trim().toUpperCase() !== 'DELETE') return;
    snapshotState('Purge Task Permanently');
    setTasks((prev) => prev.filter((t) => t.id !== taskId));
    setPurgeConfirmId(null);
    setPurgeInputText('');
  };

  const handleFileSelect = (e) => {
    const file = e.target.files?.[0];
    if (!file) return;

    const reader = new FileReader();
    reader.onload = (ev) => {
      try {
        const parsed = JSON.parse(ev.target.result);
        if (parsed.tasks && Array.isArray(parsed.tasks)) {
          const cleanTasks = parsed.tasks.map(sanitizeTask);
          setImportFilePayload({ ...parsed, tasks: cleanTasks });
          setSelectedImportTaskIds(new Set(cleanTasks.map((t) => t.id)));
        }
      } catch (err) {
        console.error('Invalid backup JSON', err);
      }
    };
    reader.readAsText(file);
    e.target.value = '';
  };

  const handleExecuteSelectiveImport = (destination = 'active') => {
    if (!importFilePayload) return;
    snapshotState(`Selective Track Import (${destination})`);

    const tasksToBringIn = importFilePayload.tasks
      .filter((t) => selectedImportTaskIds.has(t.id))
      .map((t) => ({
        ...t,
        id: generateSafeId('imp'),
        isArchived: destination === 'vault',
      }));

    setTasks((prev) => [...prev, ...tasksToBringIn]);
    setImportFilePayload(null);
  };

  const selectedTask = activeTasks.find((t) => t.id === selectedTaskId);
  const todayStatus = selectedTask ? (selectedTask.matrix?.[todayIso] || 0) : 0;
  const currentDayInfo = timelineDays[todayIndex] || {
    weekNum: 39,
    dayName: 'TODAY',
    dateStr: todayIso,
    isSunday: false,
  };

  // Keyboard Shortcuts (Enter = Start/Stop, Del = Archive, Ctrl+Z = Undo)
  useEffect(() => {
    function handleKeyDown(e) {
      const tagName = document.activeElement?.tagName;
      if (tagName === 'INPUT' || tagName === 'TEXTAREA') return;

      if (!selectedTaskId) return;

      if ((e.ctrlKey || e.metaKey) && e.key.toLowerCase() === 'z') {
        e.preventDefault();
        setTasks((prev) => executeMatrixAction(prev, selectedTaskId, 'UNDO', todayIso));
        return;
      }

      if (e.key === 'Enter') {
        e.preventDefault();
        if (todayStatus === 2) {
          setTasks((prev) => executeMatrixAction(prev, selectedTaskId, 'STOP', todayIso));
        } else if (todayStatus === 0) {
          setTasks((prev) => executeMatrixAction(prev, selectedTaskId, 'START', todayIso));
        }
        return;
      }

      if (e.key === 'Delete' || e.key === 'Backspace') {
        e.preventDefault();
        handleArchiveTask(selectedTaskId);
      }
    }

    window.addEventListener('keydown', handleKeyDown);
    return () => window.removeEventListener('keydown', handleKeyDown);
  }, [selectedTaskId, todayStatus, todayIso, handleArchiveTask]);

  return (
    <div className="flex flex-col h-screen w-screen bg-[#070707] text-white font-mono select-none overflow-hidden relative">
      <input type="file" ref={fileInputRef} onChange={handleFileSelect} accept=".json" className="hidden" />

      {/* TOP HEADER */}
      <Header
        currentDayInfo={currentDayInfo}
        onCenterWeek={() => centerCurrentWeek(true)}
        onExportBackup={() => {
          const dataStr = 'data:text/json;charset=utf-8,' + encodeURIComponent(JSON.stringify({ tasks, habits }));
          const a = document.createElement('a');
          a.setAttribute('href', dataStr);
          a.setAttribute('download', `flow_tracker_backup_${Date.now()}.json`);
          document.body.appendChild(a);
          a.click();
          a.remove();
        }}
        onTriggerFileInput={() => fileInputRef.current?.click()}
        onOpenRevisions={() => setRevisionsModalOpen(true)}
        onOpenVault={() => setVaultOpen(true)}
        archivedCount={archivedTasks.length}
        onOpenTaskModal={() => {
          setTaskForm({ title: '', hex: '#34d399', position: activeTasks.length + 1 });
          setTaskModalOpen(true);
        }}
      />

      {/* TIMELINE STREAM */}
      <TimelineStream
        scrollRef={scrollContainerRef}
        timelineDays={timelineDays}
        todayIso={todayIso}
        tasks={activeTasks}
        habits={habits}
        selectedTaskId={selectedTaskId}
        onSelectTask={setSelectedTaskId}
        onMoveTask={(taskId, direction) => {
          snapshotState(`Reorder Task: ${direction}`);
          setTasks((prev) => moveTaskPosition(prev, taskId, direction));
        }}
        onHabitClick={(habitId, targetIso) => setHabits((prev) => toggleHabitToday(prev, habitId, targetIso))}
        onDeleteHabit={(e, habitId) => {
          e.stopPropagation();
          snapshotState('Delete Habit');
          setHabits((prev) => prev.filter((h) => h.id !== habitId));
        }}
        onOpenHabitModal={() => {
          setHabitForm({ label: '', hex: '#22d3ee', position: habits.length + 1 });
          setHabitModalOpen(true);
        }}
        onColorChange={(taskId, hex) => setTasks((prev) => updateTaskColor(prev, taskId, hex))}
        editingItemId={editingItemId}
        editValue={editValue}
        setEditValue={setEditValue}
        onStartRename={(id, title) => {
          setEditingItemId(id);
          setEditValue(title);
        }}
        onSaveRename={(type, id) => {
          if (type === 'task') setTasks((prev) => renameTask(prev, id, editValue));
          setEditingItemId(null);
        }}
        onRenameHabit={(id, newLabel) => setTasks((prev) => {
          setHabits((prevH) => renameHabit(prevH, id, newLabel));
          return prev;
        })}
        onOpenHistory={setHistoryModalTask}
      />

      {/* FOOTER CONTROLS */}
      <FooterControls
        selectedTask={selectedTask}
        todayStatus={todayStatus}
        onColorChange={(taskId, hex) => setTasks((prev) => updateTaskColor(prev, taskId, hex))}
        onArchiveTask={handleArchiveTask}
        onAction={(action) => {
          if (!selectedTaskId) return;
          setTasks((prev) => executeMatrixAction(prev, selectedTaskId, action, todayIso));
        }}
      />

      {/* MODALS */}
      <TaskModal
        isOpen={taskModalOpen}
        onClose={() => setTaskModalOpen(false)}
        onSubmit={handleCreateTask}
        form={taskForm}
        setForm={setTaskForm}
        maxPosition={activeTasks.length + 1}
      />

      <HabitModal
        isOpen={habitModalOpen}
        onClose={() => setHabitModalOpen(false)}
        onSubmit={handleCreateHabit}
        form={habitForm}
        setForm={setHabitForm}
        maxPosition={habits.length + 1}
      />

      <NameHistoryModal task={historyModalTask} onClose={() => setHistoryModalTask(null)} />

      <ImportInspectorModal
        payload={importFilePayload}
        selectedIds={selectedImportTaskIds}
        onToggleSelection={(taskId) => {
          setSelectedImportTaskIds((prev) => {
            const next = new Set(prev);
            if (next.has(taskId)) next.delete(taskId);
            else next.add(taskId);
            return next;
          });
        }}
        onExecuteImport={handleExecuteSelectiveImport}
        onClose={() => setImportFilePayload(null)}
      />

      <RevisionsModal
        isOpen={revisionsModalOpen}
        onClose={() => setRevisionsModalOpen(false)}
        revisions={revisions}
        onRollback={(rev) => {
          snapshotState('Rollback to Snapshot');
          setTasks(rev.tasks.map(sanitizeTask));
          setHabits(rev.habits.map(sanitizeHabit));
          setRevisionsModalOpen(false);
        }}
      />

      <VaultDrawer
        isOpen={vaultOpen}
        onClose={() => setVaultOpen(false)}
        archivedTasks={archivedTasks}
        onRestoreTask={handleRestoreTask}
        purgeConfirmId={purgeConfirmId}
        setPurgeConfirmId={setPurgeConfirmId}
        purgeInputText={purgeInputText}
        setPurgeInputText={setPurgeInputText}
        onHardPurgeTask={handleHardPurgeTask}
      />
    </div>
  );
}

export const ROW_HEIGHT = 72;
export const TRACK_WIDTH = 44;
export const COLUMN_WIDTH = 140;
export const COLUMN_GAP = 40;
export const CANVAS_PAD_X = 48;
export const CANVAS_PAD_Y = 24;

export const getColLeft = (idx) => CANVAS_PAD_X + idx * (COLUMN_WIDTH + COLUMN_GAP);

const DAY_NAMES = ['SUN', 'MON', 'TUE', 'WED', 'THU', 'FRI', 'SAT'];

// Format any date object into local YYYY-MM-DD
export function formatISODate(date) {
  const y = date.getFullYear();
  const m = String(date.getMonth() + 1).padStart(2, '0');
  const d = String(date.getDate()).padStart(2, '0');
  return `${y}-${m}-${d}`;
}

// Get standard ISO week number (1 - 53)
export function getWeekNumber(d) {
  const date = new Date(Date.UTC(d.getFullYear(), d.getMonth(), d.getDate()));
  const dayNum = date.getUTCDay() || 7;
  date.setUTCDate(date.getUTCDate() + 4 - dayNum);
  const yearStart = new Date(Date.UTC(date.getUTCFullYear(), 0, 1));
  return Math.ceil(((date - yearStart) / 86400000 + 1) / 7);
}

// Dynamic timeline generator that scans task data and ensures at least 90 days past + future
export function generateTimelineDays(centerDate = new Date(), tasks = [], habits = []) {
  // Find earliest date recorded in any task or habit
  let earliestTime = new Date(centerDate).getTime() - (90 * 86400000); // Minimum 90 days (3 months) past

  const scanDates = (dateMap) => {
    if (!dateMap || typeof dateMap !== 'object') return;
    Object.keys(dateMap).forEach((iso) => {
      const t = new Date(iso).getTime();
      if (!isNaN(t) && t < earliestTime) {
        earliestTime = t;
      }
    });
  };

  tasks.forEach((t) => scanDates(t.matrix));
  habits.forEach((h) => scanDates(h.data));

  // Align start to the Sunday of that earliest week (with an extra week of padding)
  const startDate = new Date(earliestTime);
  startDate.setDate(startDate.getDate() - startDate.getDay() - 7);
  startDate.setHours(0, 0, 0, 0);

  // Extend 14 days into the future past centerDate
  const endDate = new Date(centerDate);
  endDate.setDate(endDate.getDate() + 14);
  endDate.setHours(0, 0, 0, 0);

  const days = [];
  const cur = new Date(startDate);
  let idx = 0;

  while (cur <= endDate) {
    const iso = formatISODate(cur);
    const dayName = DAY_NAMES[cur.getDay()];
    const dateStr = `${String(cur.getDate()).padStart(2, '0')}/${String(cur.getMonth() + 1).padStart(2, '0')}`;
    const weekNum = getWeekNumber(cur);
    const isSunday = cur.getDay() === 0;

    days.push({
      dayIndex: idx++,
      isoDate: iso,
      dayName,
      dateStr,
      weekNum,
      isSunday,
    });

    cur.setDate(cur.getDate() + 1);
  }

  return days;
}

const STORAGE_TASKS_KEY = 'flow_tracker_tasks_prod';
const STORAGE_HABITS_KEY = 'flow_tracker_habits_prod';
const STORAGE_REVISIONS_KEY = 'flow_tracker_revisions_prod';

export function generateSafeId(prefix = 'id') {
  if (typeof crypto !== 'undefined' && crypto.randomUUID) {
    return `${prefix}-${crypto.randomUUID()}`;
  }
  return `${prefix}-${Date.now()}-${Math.random().toString(36).substring(2, 9)}`;
}

// Defensive sanitization: converts legacy array matrices to permanent ISO Date maps
export function sanitizeTask(t) {
  let matrix = {};
  if (t && typeof t.matrix === 'object' && !Array.isArray(t.matrix)) {
    matrix = { ...t.matrix };
  }

  return {
    ...t,
    id: t.id || generateSafeId('col'),
    title: t.title || 'Untitled',
    hex: t.hex || '#34d399',
    streak: typeof t.streak === 'number' ? t.streak : 0,
    isArchived: Boolean(t.isArchived),
    nameHistory: Array.isArray(t.nameHistory) ? t.nameHistory : [{ title: t.title || 'Init', at: formatISODate(new Date()) }],
    matrix, // Date-keyed map: { "2026-09-23": 1 }
  };
}

export function sanitizeHabit(h) {
  let data = {};
  if (h && typeof h.data === 'object' && !Array.isArray(h.data)) {
    data = { ...h.data };
  }

  return {
    ...h,
    id: h.id || generateSafeId('habit'),
    label: (h.label || 'HABIT').substring(0, 8).toUpperCase(),
    hex: h.hex || '#22d3ee',
    nameHistory: Array.isArray(h.nameHistory) ? h.nameHistory : [{ title: h.label || 'Init', at: formatISODate(new Date()) }],
    data, // Date-keyed map: { "2026-09-23": 1 }
  };
}

export function persistState(tasks, habits, revisions = []) {
  try {
    localStorage.setItem(STORAGE_TASKS_KEY, JSON.stringify(tasks));
    localStorage.setItem(STORAGE_HABITS_KEY, JSON.stringify(habits));
    if (revisions.length) {
      localStorage.setItem(STORAGE_REVISIONS_KEY, JSON.stringify(revisions.slice(-20)));
    }
  } catch (e) {
    console.error('Persistence failed', e);
  }
}

// Parses contiguous habit blocks based on the timeline slice
export function parseHabitBlocks(habitData, timelineDays) {
  const blocks = [];
  let start = null;
  let len = 0;

  for (let i = 0; i < timelineDays.length; i++) {
    const iso = timelineDays[i].isoDate;
    const isCompleted = habitData[iso] === 1;

    if (isCompleted) {
      if (start === null) start = i;
      len++;
    } else {
      if (start !== null) {
        blocks.push({ start, days: len });
        start = null;
        len = 0;
      }
    }
  }
  if (start !== null) {
    blocks.push({ start, days: len });
  }
  return blocks;
}

// Calendar-week streak parser:
// Connects blocks across weeks as long as there is no completely dead calendar week in between.
export function parseTaskMatrix(task, timelineDays) {
  const blocks = [];
  let blockStart = null;
  let blockLen = 0;
  let hasActive = false;

  for (let i = 0; i < timelineDays.length; i++) {
    const iso = timelineDays[i].isoDate;
    const val = task.matrix[iso] || 0;

    if (val > 0) {
      if (blockStart === null) blockStart = i;
      blockLen++;
      if (val === 2) hasActive = true;
    } else {
      if (blockStart !== null) {
        blocks.push({
          start: blockStart,
          days: blockLen,
          isActive: hasActive,
        });
        blockStart = null;
        blockLen = 0;
        hasActive = false;
      }
    }
  }

  if (blockStart !== null) {
    blocks.push({
      start: blockStart,
      days: blockLen,
      isActive: hasActive,
    });
  }

  const cords = [];

  if (blocks.length > 0) {
    for (let b = 0; b < blocks.length - 1; b++) {
      const prevBlock = blocks[b];
      const nextBlock = blocks[b + 1];

      const prevEnd = prevBlock.start + prevBlock.days;
      const nextStart = nextBlock.start;
      const gapLength = nextStart - prevEnd;

      if (gapLength > 0) {
        // Look up the exact dates touched at both ends of the gap
        const lastActiveIndex = prevEnd - 1;
        const nextActiveIndex = nextStart;

        const lastActiveDate = new Date(timelineDays[lastActiveIndex].isoDate);
        const nextActiveDate = new Date(timelineDays[nextActiveIndex].isoDate);

        // Calculate the difference in calendar weeks (Sunday to Saturday boundaries)
        const msPerDay = 86400000;
        // Align both dates to their week's Sunday (start of calendar week)
        const lastSunday = new Date(lastActiveDate);
        lastSunday.setDate(lastActiveDate.getDate() - lastActiveDate.getDay());
        lastSunday.setHours(0, 0, 0, 0);

        const nextSunday = new Date(nextActiveDate);
        nextSunday.setDate(nextActiveDate.getDate() - nextActiveDate.getDay());
        nextSunday.setHours(0, 0, 0, 0);

        const weekDiff = Math.round((nextSunday.getTime() - lastSunday.getTime()) / (7 * msPerDay));

        // Connect if:
        // - In the same week (weekDiff === 0)
        // - In consecutive weeks (weekDiff === 1, e.g. Sunday to next Monday or Saturday)
        // Breaks ONLY if at least one whole week was skipped (weekDiff >= 2)
        if (weekDiff <= 1) {
          cords.push({ startY: prevEnd, endY: nextStart });
        }
      }
    }
  }

  return { blocks, cords };
}

// Action executor keyed directly by real ISO date
export function executeMatrixAction(tasks, taskId, action, todayIso) {
  return tasks.map((task) => {
    if (task.id !== taskId) return task;

    const nextMatrix = { ...(task.matrix || {}) };
    const currentVal = nextMatrix[todayIso] || 0;

    if (action === 'START') {
      if (currentVal === 0) nextMatrix[todayIso] = 2;
    } else if (action === 'STOP') {
      if (currentVal === 2) nextMatrix[todayIso] = 1;
    } else if (action === 'UNDO') {
      delete nextMatrix[todayIso];
    }

    return { ...task, matrix: nextMatrix };
  });
}

// Habit toggle keyed directly by real ISO date
export function toggleHabitToday(habits, habitId, targetIso) {
  return habits.map((h) => {
    if (h.id !== habitId) return h;
    const nextData = { ...(h.data || {}) };
    if (nextData[targetIso] === 1) {
      delete nextData[targetIso];
    } else {
      nextData[targetIso] = 1;
    }
    return { ...h, data: nextData };
  });
}

export function renameTask(tasks, taskId, newTitle) {
  const trimmed = newTitle.trim();
  return tasks.map((t) => {
    if (t.id !== taskId || !trimmed || t.title === trimmed) return t;
    const history = t.nameHistory || [];
    return {
      ...t,
      title: trimmed,
      nameHistory: [...history, { title: trimmed, at: formatISODate(new Date()) }]
    };
  });
}

export function renameHabit(habits, habitId, newLabel) {
  const trimmed = newLabel.trim().substring(0, 8).toUpperCase();
  return habits.map((h) => {
    if (h.id !== habitId || !trimmed || h.label === trimmed) return h;
    const history = h.nameHistory || [];
    return {
      ...h,
      label: trimmed,
      nameHistory: [...history, { title: trimmed, at: formatISODate(new Date()) }]
    };
  });
}

export function updateTaskColor(tasks, taskId, newHex) {
  return tasks.map((t) => (t.id === taskId ? { ...t, hex: newHex } : t));
}

export function insertNewTask(tasks, { title, hex, position }) {
  const newTask = {
    id: generateSafeId('col'),
    title: title.trim().substring(0, 16),
    hex: hex || '#34d399',
    streak: 0,
    isArchived: false,
    nameHistory: [{ title: title.trim(), at: formatISODate(new Date()) }],
    matrix: {},
  };

  const parsedPos = parseInt(position, 10);
  const pos = Number.isFinite(parsedPos) ? parsedPos : 1;

  // Filter only visible active tasks to calculate true visual column slot
  const activeOnly = tasks.filter((t) => !t.isArchived);
  const targetActiveIndex = Math.max(0, Math.min(pos - 1, activeOnly.length));

  if (targetActiveIndex >= activeOnly.length) {
    // Append after all current tasks
    return [...tasks, newTask];
  } else {
    // Place right before the active task currently at that slot
    const referenceTask = activeOnly[targetActiveIndex];
    const realIndex = tasks.findIndex((t) => t.id === referenceTask.id);
    const copy = [...tasks];
    copy.splice(realIndex, 0, newTask);
    return copy;
  }
}

export function insertNewHabit(habits, { label, hex, position }) {
  const newHabit = {
    id: generateSafeId('habit'),
    label: label.trim().substring(0, 16).toUpperCase(),
    hex: hex || '#22d3ee',
    nameHistory: [{ title: label.trim().substring(0, 16).toUpperCase(), at: formatISODate(new Date()) }],
    data: {},
  };

  const parsedPos = parseInt(position, 10);
  const pos = Number.isFinite(parsedPos) ? parsedPos : 1;
  const targetIndex = Math.max(0, Math.min(pos - 1, habits.length));

  const copy = [...habits];
  copy.splice(targetIndex, 0, newHabit);
  return copy;
}
export function moveTaskPosition(tasks, taskId, direction) {
  const activeTasks = tasks.filter((t) => !t.isArchived);
  const fromIndex = activeTasks.findIndex((t) => t.id === taskId);
  if (fromIndex === -1) return tasks;

  const toIndex = direction === 'left' ? fromIndex - 1 : fromIndex + 1;
  if (toIndex < 0 || toIndex >= activeTasks.length) return tasks;

  // Swap target elements in activeTasks
  const targetTask = activeTasks[toIndex];
  
  // Map back to main tasks array
  const fullFrom = tasks.findIndex((t) => t.id === taskId);
  const fullTo = tasks.findIndex((t) => t.id === targetTask.id);

  const copy = [...tasks];
  const temp = copy[fullFrom];
  copy[fullFrom] = copy[fullTo];
  copy[fullTo] = temp;

  return copy;
}

                            ? 'border-white/15' 
                            : 'border-white/5'
                    }`} 
                  />
                </div>
              );
            })}
          </div>

          <div className="absolute inset-0 z-10">
            {tasks.map((task, idx) => {

              const { blocks, cords } = parseTaskMatrix(task, timelineDays);

              return (
                <div
                  key={`${task.id}-${idx}`}
                  style={{ left: `${getColLeft(idx)}px`, width: `${COLUMN_WIDTH}px` }}
                  className={`absolute h-full flex justify-center cursor-pointer transition-colors ${
                    isSelected ? 'bg-white/[0.04]' : 'hover:bg-white/[0.02]'
                  }`}
                  onClick={() => onSelectTask(task.id)}
                >
                  <div className="absolute inset-y-0 w-[1px] bg-white/5 pointer-events-none" />

                  {/* Umbilical Cords */}
                  {cords.map((cord, cIdx) => {

                    const height = (cord.endY - cord.startY) * ROW_HEIGHT;

                        style={{ 
                          top: `${top}px`, 
                          height: `${height}px`,

                          boxShadow: `0 0 12px ${task.hex}80`
                        }}
                        className="absolute w-[3px] opacity-70 pointer-events-none"
                      />
                    );
                  })}

                  {/* Fluid Block Trackers with Neon Luminescence */}
                  <div className="relative flex justify-center" style={{ width: `${TRACK_WIDTH}px` }}>
                    {blocks.map((block, bIdx) => {
                      const top = CANVAS_PAD_Y + block.start * ROW_HEIGHT;
                      const height = block.days * ROW_HEIGHT;
                      const isGlowing = task.streak > 3 || block.isActive;
